   194→// CONFETTI PARTICLE SYSTEM
   195→// ============================================
   196→function Confetti({ confettiKey }: { confettiKey: number }) {
   197→  const [particles, setParticles] = useState<
   198→    { id: number; x: number; y: number; color: string; size: number; rotation: number }[]
   199→  >([]);
   200→
   201→  useEffect(() => {
   202→    if (confettiKey === 0) return;
   203→    const colors = ["#E5B83C", "#E07A2C", "#25D366", "#B23A1A", "#F8E5B0", "#2E7D32"];
   204→    const newParticles = Array.from({ length: 25 }, (_, i) => ({
   205→      id: Date.now() + i,
   206→      x: 30 + Math.random() * 40,
   207→      y: 50 + Math.random() * 20,
   208→      color: colors[Math.floor(Math.random() * colors.length)],
   209→      size: 6 + Math.random() * 8,
   210→      rotation: Math.random() * 360,
   211→    }));
   212→    setParticles(newParticles);
   213→    const timer = setTimeout(() => setParticles([]), 1500);
   214→    return () => clearTimeout(timer);
   215→  }, [confettiKey]);
   216→
   217→  return (
   218→    <div className="fixed inset-0 pointer-events-none z-[9999] overflow-hidden">
   219→      <AnimatePresence>
   220→        {particles.map((p) => (
   221→          <motion.div
   222→            key={p.id}
   223→            initial={{ x: `${p.x}vw`, y: `${p.y}vh`, rotate: p.rotation, opacity: 1 }}
   224→            animate={{ y: "110vh", rotate: p.rotation + 720, opacity: 0 }}
   225→            exit={{ opacity: 0 }}
   226→            transition={{ duration: 1.2 + Math.random() * 0.5, ease: "easeIn" }}
   227→            className="absolute rounded-sm"
   228→            style={{
   229→              width: p.size,
   230→              height: p.size,
   231→              backgroundColor: p.color,
   232→            }}
   233→          />
   234→        ))}
   235→      </AnimatePresence>
   236→    </div>
   237→  );
   238→}

   240→// ============================================
   241→// SMOKE CANVAS
   242→// ============================================
   243→function SmokeCanvas() {
   244→  const canvasRef = useRef<HTMLCanvasElement>(null);
   245→
   246→  useEffect(() => {
   247→    const canvas = canvasRef.current;
   248→    if (!canvas) return;
   249→    const ctx = canvas.getContext("2d");
   250→    if (!ctx) return;
   251→
   252→    let w = (canvas.width = window.innerWidth);
   253→    let h = (canvas.height = window.innerHeight);
   254→    let animationId: number;
   255→
   256→    const particles = Array.from({ length: 30 }, () => ({
   257→      x: Math.random() * w,
   258→      y: Math.random() * h,
   259→      r: 40 + Math.random() * 80,
   260→      vx: (Math.random() - 0.5) * 0.15,
   261→      vy: -(Math.random() * 0.4 + 0.08),
   262→    }));
   263→
   264→    const animate = () => {
   265→      ctx.clearRect(0, 0, w, h);
   266→      particles.forEach((p) => {
   267→        p.x += p.vx;
   268→        p.y += p.vy;
   269→        if (p.y < -120) p.y = h + 60;
   270→        if (p.x < -120) p.x = w + 60;
   271→        if (p.x > w + 120) p.x = -120;
   272→        const grad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.r);
   273→        grad.addColorStop(0, "rgba(210,90,30,0.035)");
   274→        grad.addColorStop(1, "rgba(0,0,0,0)");
   275→        ctx.fillStyle = grad;
   276→        ctx.beginPath();
   277→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   278→        ctx.fill();
   279→      });
   280→      animationId = requestAnimationFrame(animate);
   281→    };
   282→    animate();
   283→
   284→    const handleResize = () => {
   285→      w = canvas.width = window.innerWidth;
   286→      h = canvas.height = window.innerHeight;
   287→    };
   288→    window.addEventListener("resize", handleResize);
   289→    return () => {
   290→      cancelAnimationFrame(animationId);
   291→      window.removeEventListener("resize", handleResize);
   292→    };
   293→  }, []);
   294→
   295→  return <canvas ref={canvasRef} className="fixed inset-0 pointer-events-none z-[1] opacity-60" />;
   296→}

   708→function ProductsSection({ onItemAdd, productRefs }: { onItemAdd?: () => void; productRefs?: React.RefObject<(HTMLDivElement | null)[]> }) {
   709→  // Fetch live products from /api/products. Falls back to hardcoded PRODUCTS
   710→  // if the API is unreachable (e.g. Supabase down) so the page still renders.
   711→  const [products, setProducts] = useState<Product[]>(PRODUCTS);
   712→  const [loading, setLoading] = useState(true);
   713→
   714→  useEffect(() => {
   715→    let cancelled = false;
   716→    (async () => {
   717→      try {
   718→        const res = await fetch("/api/products");
   719→        if (!res.ok) return;
   720→        const data = await res.json();
   721→        if (!cancelled && Array.isArray(data.products) && data.products.length > 0) {
   722→          setProducts(data.products);
   723→        }
   724→      } catch {
   725→        // Silent — fallback to hardcoded PRODUCTS already in state
   726→      } finally {
   727→        if (!cancelled) setLoading(false);
   728→      }
   729→    })();
   730→    return () => { cancelled = true; };
   731→  }, []);
   732→
   733→  return (
   734→    <section id="products" className="py-16 md:py-24 px-4 md:px-[6%] relative z-20">
   735→      <motion.div variants={fadeUp} initial="hidden" whileInView="visible" viewport={{ once: true }} className="text-center">
   736→        <p className="text-[0.6rem] tracking-[0.4em] uppercase text-[#E07A2C] mb-3">THE MENU</p>
   737→        <h2 className="font-['Cormorant_Garamond'] text-[2.5rem] md:text-[4.5rem] font-light leading-tight mb-6">
   738→          Choose your <em className="italic text-[#E5B83C] font-semibold">Biltong</em>
   739→        </h2>
   740→      </motion.div>
   741→
   742→      <motion.div variants={staggerContainer} initial="hidden" whileInView="visible" viewport={{ once: true, margin: "-30px" }}
   743→        className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-6 max-w-[1400px] mx-auto">
   744→        {loading ? (
   745→          // Skeleton placeholders while fetching
   746→          Array.from({ length: 4 }).map((_, i) => (
   747→            <div key={`skeleton-${i}`} className="bg-white/4 border border-white/8 rounded-2xl p-6 animate-pulse">
   748→              <div className="w-full h-40 md:h-52 bg-white/6 rounded-xl mb-3" />
   749→              <div className="h-5 bg-white/6 rounded mb-2 w-3/4" />
   750→              <div className="h-3 bg-white/4 rounded mb-4 w-full" />
   751→              <div className="h-8 bg-white/6 rounded w-1/2" />
   752→            </div>
   753→          ))
   754→        ) : (
   755→          products.map((prod, i) => (
   756→            <ProductCard key={prod.id} product={prod} index={i} onAdd={onItemAdd} cardRef={productRefs ? { current: productRefs.current?.[i] ?? null } as React.RefObject<HTMLDivElement | null> : undefined} />
   757→          ))
   758→        )}
   759→      </motion.div>
   760→    </section>
   761→  );
   762→}
