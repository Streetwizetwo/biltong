    86→// ============================================
    87→// AUTH HELPERS (cookie-based, no token storage)
    88→// ============================================
    89→// The session token is stored in an httpOnly cookie set by the server.
    90→// JavaScript can NEVER read it — defeats XSS token theft.
    91→// All API calls include credentials: "include" so the cookie is sent.
    92→
    93→async function checkAuth(): Promise<boolean> {
    94→  try {
    95→    const res = await fetch("/api/admin/auth", { method: "GET", credentials: "include" });
    96→    if (!res.ok) return false;
    97→    const data = await res.json();
    98→    return data.valid === true;
    99→  } catch {
   100→    return false;
   101→  }
   102→}
   103→
   104→async function logout(): Promise<void> {
   105→  await fetch("/api/admin/auth", { method: "DELETE", credentials: "include" });
   106→}

   110→// ============================================
   111→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   112→  const [password, setPassword] = useState("");
   113→  const [showPassword, setShowPassword] = useState(false);
   114→  const [error, setError] = useState("");
   115→  const [shaking, setShaking] = useState(false);
   116→  const [loggingIn, setLoggingIn] = useState(false);
   117→
   118→  const handleSubmit = async (e: React.FormEvent) => {
   119→    e.preventDefault();
   120→    setLoggingIn(true);
   121→    setError("");
   122→
   123→    try {
   124→      // Send password to server — server sets httpOnly cookie on success
   125→      const res = await fetch("/api/admin/auth", {
   126→        method: "POST",
   127→        headers: { "Content-Type": "application/json" },
   128→        credentials: "include",
   129→        body: JSON.stringify({ password }),
   130→      });
   131→
   132→      const data = await res.json();
   133→
   134→      if (res.ok && data.success) {
   135→        // Cookie is set by the server — no token to store in JS
   136→        onLogin();
   137→      } else if (res.status === 429) {
   138→        setError(data.error || "Too many attempts. Please wait.");
   139→        setShaking(true);
   140→        setTimeout(() => setShaking(false), 500);
   141→      } else {
   142→        setError(data.error || "Wrong password. Try again.");
   143→        setShaking(true);
   144→        setTimeout(() => setShaking(false), 500);
   145→      }
   146→    } catch {
   147→      setError("Connection error. Please try again.");
   148→    } finally {
   149→      setLoggingIn(false);
   150→    }
   151→  };
   152→
   153→  return (
   154→    <div className="min-h-screen bg-gradient-to-br from-[#0A0301] via-[#1A0A04] to-[#0A0301] flex items-center justify-center px-4">
   155→      <motion.div
   156→        initial={{ opacity: 0, y: 30 }}
   157→        animate={{ opacity: 1, y: shaking ? [0, -10, 10, -5, 5, 0] : 0 }}
   158→        transition={{ duration: 0.5 }}
   159→        className="w-full max-w-sm"
   160→      >
   161→        <div className="text-center mb-8">
   162→          <div className="w-20 h-20 bg-[#E5B83C]/15 border-2 border-[#E5B83C] rounded-full flex items-center justify-center mx-auto mb-4">
   163→            <Lock className="w-8 h-8 text-[#E5B83C]" />
   164→          </div>
   165→          <h1 className="font-['Cormorant_Garamond'] text-3xl text-[#FEF3DF]">Admin Dashboard</h1>
   166→          <p className="text-sm text-[#FEF3DF]/50 mt-1">Biltong & Bytes</p>
   167→        </div>
   168→
   169→        <form onSubmit={handleSubmit} className="space-y-4">
   170→          <div className="relative">
   171→            <input
   172→              type={showPassword ? "text" : "password"}
   173→              value={password}
   174→              onChange={(e) => { setPassword(e.target.value); setError(""); }}
   175→              placeholder="Enter admin password"
   176→              className="w-full bg-white/8 border border-[#E5B83C]/30 rounded-xl px-4 py-3.5 text-[#FEF3DF] text-sm placeholder:text-[#FEF3DF]/25 focus:outline-none focus:border-[#E5B83C] focus:bg-white/12 transition-all pr-12"
   177→              autoFocus
   178→            />
   179→            <button
   180→              type="button"
   181→              onClick={() => setShowPassword(!showPassword)}
   182→              className="absolute right-3 top-1/2 -translate-y-1/2 text-[#FEF3DF]/40 hover:text-[#E5B83C] cursor-pointer"
   183→            >
   184→              {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   185→            </button>
   186→          </div>
   187→
   188→          {error && (
   189→            <motion.p initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-red-400 text-xs text-center">
   190→              {error}
   191→            </motion.p>
   192→          )}
   193→
   194→          <button
   195→            type="submit"
   196→            disabled={loggingIn}
   197→            className={`w-full py-3.5 font-bold tracking-[0.1em] uppercase cursor-pointer rounded-xl text-sm transition-all flex items-center justify-center gap-2 ${
   198→              loggingIn ? "bg-[#E5B83C]/50 text-[#0A0301]/70 cursor-not-allowed" : "bg-[#E5B83C] text-[#0A0301] hover:bg-[#E5B83C]/90"
   199→            }`}
   200→          >
   201→            {loggingIn ? <Loader2 className="w-4 h-4 animate-spin" /> : null}
   202→            {loggingIn ? "VERIFYING..." : "LOGIN"}
   203→          </button>
   204→        </form>
   205→      </motion.div>
   206→    </div>
   207→  );
   208→}

   213→function StatsCards({ orders }: { orders: Order[] }) {
   214→  const totalRevenue = orders
   215→    .filter((o) => o.payment_status === "paid" || o.order_status === "confirmed" || o.order_status === "fulfilled")
   216→    .reduce((sum, o) => sum + o.total, 0);
   217→
   218→  const pendingCount = orders.filter((o) => o.order_status === "new" || o.order_status === "payment_initiated").length;
   219→  const fulfilledCount = orders.filter((o) => o.order_status === "fulfilled").length;
   220→  const totalOrders = orders.length;
   221→
   222→  const stats = [
   223→    { label: "Total Orders", value: totalOrders, icon: ShoppingBag, color: "text-blue-400", bg: "bg-blue-500/10" },
   224→    { label: "Revenue", value: `R${totalRevenue}`, icon: CircleDollarSign, color: "text-green-400", bg: "bg-green-500/10" },
   225→    { label: "Pending", value: pendingCount, icon: Clock, color: "text-amber-400", bg: "bg-amber-500/10" },
   226→    { label: "Fulfilled", value: fulfilledCount, icon: Package, color: "text-[#E5B83C]", bg: "bg-[#E5B83C]/10" },
   227→  ];
   228→
   229→  return (
   230→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   231→      {stats.map((stat) => (
   232→        <div key={stat.label} className={`${stat.bg} border border-white/10 rounded-xl p-4`}>
   233→          <stat.icon className={`w-5 h-5 ${stat.color} mb-2`} />
   234→          <p className="font-['Bebas_Neue'] text-2xl text-[#FEF3DF]">{stat.value}</p>
   235→          <p className="text-[0.6rem] text-[#FEF3DF]/50 tracking-wider uppercase">{stat.label}</p>
   236→        </div>
   237→      ))}
   238→    </div>
   239→  );
   240→}
